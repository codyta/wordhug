import { Card, CardGrid } from "@astrojs/starlight/components";
import WordHug from "wordhug/astro";
import Demo from "../../components/Demo.astro";
import Plain from "../../components/Plain.astro";

export const text = "Mẹo chọn đồ giúp bạn [ăn mặc] đẹp hơn mỗi ngày";

<div class="demo-row">
  <Demo label="Without wordhug" width={22}>
    <p><Plain text={text} /></p>
  </Demo>
  <Demo label="With wordhug" width={22}>
    <p><WordHug text={text} /></p>
  </Demo>
</div>

Drag the corner of either box. Without wordhug, the Vietnamese word *ăn mặc* (to dress) can be split across two lines, leaving *ăn* (to eat) on its own: the heading then reads "tips to help you eat". With brackets around it, *ăn mặc* always stays together.

```tsx
<WordHug>{"Mẹo chọn đồ giúp bạn [ăn mặc] đẹp hơn mỗi ngày"}</WordHug>
// → Mẹo chọn đồ giúp bạn <span style="display: inline-block">ăn mặc</span> đẹp hơn mỗi ngày
```
